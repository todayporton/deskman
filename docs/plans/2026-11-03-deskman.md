# deskman 実装計画

> **For Claude:** REQUIRED SUB-SKILL: Use superpowers:executing-plans to implement this plan task-by-task.

**Goal:** macOS のデスクトップに名前を付け、並べ替えや増減、ディスプレイの抜き差しがあっても名前で移動できる CLI とメニューバーアプリを作る。

**Architecture:** SwiftPM のパッケージ `deskman/`。純粋なロジックを `DeskmanCore` に置いて単体テストし、非公式 API の呼び出しは `SystemSpaceSource` と `SpaceSwitcher` に閉じ込める。1つのバイナリが、引数なしならメニューバー、引数ありなら CLI として動く。

**Tech Stack:** Swift 6、SwiftPM、XCTest、AppKit、SkyLight（非公式）、CoreGraphics

**設計書:** `docs/plans/2026-11-03-deskman-design.md`

```swift
// Sources/DeskmanCore/SpaceParser.swift
public enum SpaceParser {
    public static func parse(_ raw: [[String: Any]], activeID: Int) -> [Space] {
        var result: [Space] = []
        for display in raw {
            let displayID = display["Display Identifier"] as? String ?? ""
            let current = (display["Current Space"] as? [String: Any])?["uuid"] as? String
            for s in display["Spaces"] as? [[String: Any]] ?? [] {
                // UUID が空のデスクトップは名前を付けられないので扱わない（spacelabel の知見）
                guard let uuid = s["uuid"] as? String, !uuid.isEmpty else { continue }
                let id = (s["id64"] as? Int) ?? (s["ManagedSpaceID"] as? Int) ?? 0
                result.append(Space(
                    uuid: uuid, managedID: id, displayID: displayID,
                    index: result.count + 1,
                    // フルスクリーンでも type が 0 の場合がある（spacelabel の知見）
                    isFullscreen: (s["type"] as? Int ?? 0) != 0 || s["TileLayoutManager"] != nil,
                    isVisible: uuid == current,
                    isActive: id == activeID))
            }
        }
        return result
    }
}
```

保存は一時ファイルからの置き換え（`.atomic`）で行い、書き込み途中で壊れないようにする。

```swift
import Foundation

public struct NameStore {
    public static var defaultURL: URL {
        FileManager.default.homeDirectoryForCurrentUser
            .appendingPathComponent(".config/deskman/names.json")
    }
    public private(set) var names: [String: String]
    private let url: URL

    public init(url: URL = NameStore.defaultURL) throws {
        self.url = url
        if let data = try? Data(contentsOf: url) {
            names = try JSONDecoder().decode([String: String].self, from: data)
        } else {
            names = [:]
        }
    }
    public func uuid(forName name: String) -> String? {
        names.first { $0.value == name }?.key
    }
    public mutating func set(name: String, for uuid: String) throws {
        if name.isEmpty {
            names[uuid] = nil
        } else {
            if let owner = self.uuid(forName: name), owner != uuid {
                throw DeskmanError.duplicateName(name)
            }
            names[uuid] = name
        }
        try save()
    }
    public mutating func forget(name: String, liveUUIDs: Set<String>) throws {
        guard let uuid = uuid(forName: name) else { throw DeskmanError.nameNotFound(name) }
        if liveUUIDs.contains(uuid) { throw DeskmanError.hasLiveSpace(name) }
        names[uuid] = nil
        try save()
    }
    private func save() throws {
        try FileManager.default.createDirectory(
            at: url.deletingLastPathComponent(), withIntermediateDirectories: true)
        let enc = JSONEncoder()
        enc.outputFormatting = [.prettyPrinted, .sortedKeys]
        try enc.encode(names).write(to: url, options: .atomic)
    }
}
```

優先順位は 名前 → UUID → 番号。数字の名前が番号より優先される。

```swift
// SpaceResolver.swift
public enum SpaceResolver {
    public static func resolve(_ query: String, spaces: [Space], names: [String: String]) throws -> Space {
        if let uuid = names.first(where: { $0.value == query })?.key {
            guard let s = spaces.first(where: { $0.uuid == uuid }) else {
                throw DeskmanError.notInList(query)
            }
            return s
        }
        if let s = spaces.first(where: { $0.uuid == query }) { return s }
        if let n = Int(query), let s = spaces.first(where: { $0.index == n }) { return s }
        throw DeskmanError.nameNotFound(query)
    }
}
```

```swift
// SwitchPlan.swift
public enum SwitchPlan: Equatable {
    case alreadyThere
    /// steps が正なら右、負なら左。
    case swipe(displayID: String, steps: Int, changeDisplay: Bool)

    public static func make(_ target: Space, in spaces: [Space]) -> SwitchPlan {
        if target.isActive { return .alreadyThere }
        let visible = spaces.first { $0.displayID == target.displayID && $0.isVisible }
        let steps = target.index - (visible?.index ?? target.index)
        let activeDisplay = spaces.first(where: \.isActive)?.displayID
        return .swipe(displayID: target.displayID, steps: steps,
                      changeDisplay: activeDisplay != target.displayID)
    }
}
```

**Step 3: 実装する**

`go` は番号札を取り、実行ロックを待ち、ロック取得後に自分が最新の札でなければ何もせず成功で終わる。プロセスをまたぐので `flock` を使う。

```swift
import Foundation

public struct GoQueue {
    public static var defaultURL: URL {
        FileManager.default.homeDirectoryForCurrentUser.appendingPathComponent(".config/deskman/go-ticket")
    }
    private let url: URL
    public init(url: URL = GoQueue.defaultURL) { self.url = url }

    private func locked<T>(_ path: String, _ body: (Int32) throws -> T) throws -> T {
        try FileManager.default.createDirectory(
            at: URL(fileURLWithPath: path).deletingLastPathComponent(), withIntermediateDirectories: true)
        let fd = open(path, O_RDWR | O_CREAT, 0o644)
        guard fd >= 0 else { throw POSIXError(.EIO) }
        defer { close(fd) }
        flock(fd, LOCK_EX)
        defer { flock(fd, LOCK_UN) }
        return try body(fd)
    }
    private func read(_ fd: Int32) -> Int {
        lseek(fd, 0, SEEK_SET)
        var buf = [UInt8](repeating: 0, count: 32)
        let n = Foundation.read(fd, &buf, 32)
        return n > 0 ? Int(String(decoding: buf[0..<n], as: UTF8.self)) ?? 0 : 0
    }
    public func takeTicket() throws -> Int {
        try locked(url.path) { fd in
            let next = read(fd) + 1
            ftruncate(fd, 0); lseek(fd, 0, SEEK_SET)
            let s = Array(String(next).utf8)
            _ = write(fd, s, s.count)
            return next
        }
    }
    public func isLatest(_ ticket: Int) throws -> Bool {
        try locked(url.path) { read($0) == ticket }
    }
    public func withRunLock(_ body: () throws -> Void) throws {
        try locked(url.path + ".lock") { _ in try body() }
    }
}
```

```swift
// SpaceSwitcher.swift
import ApplicationServices
import ColorSync
import CoreGraphics
import Foundation

public enum SpaceSwitcher {
    public static func go(to target: Space) throws {
        let spaces = SystemSpaces.current()
        // 調停の待ち時間の間に並びが変わった場合に備えて取り直す
        guard let fresh = spaces.first(where: { $0.uuid == target.uuid }) else {
            throw DeskmanError.notInList(target.uuid)
        }
        guard case let .swipe(displayID, steps, changeDisplay) = SwitchPlan.make(fresh, in: spaces) else {
            return
        }
        guard AXIsProcessTrusted() else { throw DeskmanError.notTrusted }
        if changeDisplay, let point = center(ofDisplay: displayID) {
            CGWarpMouseCursorPosition(point)
        }
        postSwipes(steps)
        if changeDisplay, steps == 0, let point = center(ofDisplay: displayID) {
            click(at: point)
        }
        try confirm(target.uuid)
    }

    private static func field(_ n: UInt32) -> CGEventField { unsafeBitCast(n, to: CGEventField.self) }

    private static func postSwipes(_ steps: Int) {
        guard steps != 0, let e = CGEvent(source: nil) else { return }
        let sign: Double = steps > 0 ? 1 : -1
        e.setIntegerValueField(field(55), value: 30)
        e.setIntegerValueField(field(110), value: 23)
        e.setIntegerValueField(field(123), value: 1)
        e.setDoubleValueField(field(124), value: sign)
        e.setDoubleValueField(field(129), value: sign * 9999.0)
        for _ in 0..<abs(steps) {
            e.setIntegerValueField(field(132), value: 1)
            e.post(tap: .cgSessionEventTap)
            e.setIntegerValueField(field(132), value: 4)
            e.post(tap: .cgSessionEventTap)
        }
    }

    private static func center(ofDisplay id: String) -> CGPoint? {
        var ids = [CGDirectDisplayID](repeating: 0, count: 16)
        var n: UInt32 = 0
        guard CGGetActiveDisplayList(16, &ids, &n) == .success else { return nil }
        for d in ids.prefix(Int(n)) {
            guard let cf = CGDisplayCreateUUIDFromDisplayID(d)?.takeRetainedValue() else { continue }
            let s = CFUUIDCreateString(nil, cf) as String
            if s.caseInsensitiveCompare(id) == .orderedSame {
                let b = CGDisplayBounds(d)
                return CGPoint(x: b.midX, y: b.midY)
            }
        }
        return nil
    }

    /// 移動先のディスプレイにキーボードフォーカスを移す。yabai と同じ手法で、
    /// カーソル位置のウィンドウにもクリックが届くため、Task 9 で影響を確認する。
    private static func click(at p: CGPoint) {
        for t in [CGEventType.leftMouseDown, .leftMouseUp] {
            CGEvent(mouseEventSource: nil, mouseType: t, mouseCursorPosition: p, mouseButton: .left)?
                .post(tap: .cghidEventTap)
        }
    }

    private static func confirm(_ uuid: String) throws {
        let deadline = Date().addingTimeInterval(1)
        while Date() < deadline {
            if SystemSpaces.current().contains(where: { $0.uuid == uuid && $0.isVisible }) { return }
            Thread.sleep(forTimeInterval: 0.03)
        }
        throw DeskmanError.switchNotConfirmed
    }
}
```

```swift
public struct SpaceRow: Codable, Equatable {
    public let uuid: String
    public let name: String?
    public let index: Int?
    public let display: String?
    public let visible: Bool
    public let active: Bool
    public let fullscreen: Bool
    public let present: Bool

    public static func make(spaces: [Space], names: [String: String], includeAbsent: Bool) -> [SpaceRow] {
        var rows = spaces.map {
            SpaceRow(uuid: $0.uuid, name: names[$0.uuid], index: $0.index, display: $0.displayID,
                     visible: $0.isVisible, active: $0.isActive, fullscreen: $0.isFullscreen, present: true)
        }
        if includeAbsent {
            let live = Set(spaces.map(\.uuid))
            for (uuid, name) in names.sorted(by: { $0.value < $1.value }) where !live.contains(uuid) {
                rows.append(SpaceRow(uuid: uuid, name: name, index: nil, display: nil, visible: false,
                                     active: false, fullscreen: false, present: false))
            }
        }
        return rows
    }
```

```swift
// Sources/deskman/CLI.swift
import Foundation
import DeskmanCore

func json<T: Encodable>(_ v: T) throws -> String {
    let e = JSONEncoder()
    e.outputFormatting = [.prettyPrinted, .sortedKeys, .withoutEscapingSlashes]
    return String(decoding: try e.encode(v), as: UTF8.self)
}

func runCLI(_ args: [String]) throws {
    var args = args
    let command = args.removeFirst()
    func flag(_ f: String) -> Bool {
        if let i = args.firstIndex(of: f) { args.remove(at: i); return true }
        return false
    }
    var store = try NameStore()
    let spaces = SystemSpaces.current()

    switch command {
    case "list":
        let asJSON = flag("--json")
        let rows = SpaceRow.make(spaces: spaces, names: store.names, includeAbsent: flag("--all"))
        if asJSON { print(try json(rows)); return }
        for r in rows {
            let mark = r.active ? "*" : (r.visible ? "+" : " ")
            let index = r.index.map(String.init) ?? "-"
            let suffix = r.present ? (r.fullscreen ? "（フルスクリーン）" : "") : "（一覧になし）"
            print("\(mark) \(index)\t\(r.name ?? "-")\t\(r.display ?? "-")\t\(r.uuid)\(suffix)")
        }
    case "current":
        let asJSON = flag("--json")
        guard let s = spaces.first(where: \.isActive) else { throw DeskmanError.nameNotFound("current") }
        let row = SpaceRow.make(spaces: [s], names: store.names, includeAbsent: false)[0]
        print(asJSON ? try json(row) : (row.name ?? row.uuid))
    case "go":
        guard let q = args.first else { usage() }
        // 先に解決して、間違った名前なら順番待ちせずに失敗させる
        _ = try SpaceResolver.resolve(q, spaces: spaces, names: store.names)
        let queue = GoQueue()
        let ticket = try queue.takeTicket()
        try queue.withRunLock {
            guard try queue.isLatest(ticket) else { return }
            let fresh = SystemSpaces.current()
            try SpaceSwitcher.go(to: SpaceResolver.resolve(q, spaces: fresh, names: store.names))
        }
    case "rename":
        if flag("--forget") {
            guard let name = args.first else { usage() }
            try store.forget(name: name, liveUUIDs: Set(spaces.map(\.uuid)))
            return
        }
        guard args.count == 2 else { usage() }
        let target = try SpaceResolver.resolve(args[0], spaces: spaces, names: store.names)
        if target.isFullscreen { throw DeskmanError.fullscreenNotNameable }
        try store.set(name: args[1], for: target.uuid)
    default:
        usage()
    }
}

func usage() -> Never {
    FileHandle.standardError.write(Data("""
    使い方:
      deskman list [--json] [--all]
      deskman current [--json]
      deskman go <名前|番号|UUID>
      deskman rename <対象> <名前>
      deskman rename --forget <名前>

    """.utf8))
    exit(64)
}

func message(for error: DeskmanError) -> String {
    switch error {
    case .nameNotFound(let q): "見つかりません: \(q)"
    case .notInList(let q): "\(q) は今の一覧にありません。ディスプレイが外れている可能性があります。"
    case .notTrusted: "アクセシビリティ権限が必要です。システム設定 > プライバシーとセキュリティ > アクセシビリティ で許可してください。"
    case .switchNotConfirmed: "移動を確認できませんでした。"
    case .duplicateName(let n): "すでに使われている名前です: \(n)"
    case .fullscreenNotNameable: "フルスクリーンのデスクトップには名前を付けられません。"
    case .hasLiveSpace(let n): "\(n) は今の一覧にあるので消せません。解除するには rename <対象> \"\" を使ってください。"
    }
}
```

```swift
// Sources/deskman/MenuBar.swift
import AppKit
import DeskmanCore

@MainActor
final class MenuBarController: NSObject, NSMenuDelegate {
    let item = NSStatusBar.system.statusItem(withLength: NSStatusItem.variableLength)
    let menu = NSMenu()

    override init() {
        super.init()
        menu.delegate = self
        item.menu = menu
        refreshTitle()
        NSWorkspace.shared.notificationCenter.addObserver(
            self, selector: #selector(refreshTitle),
            name: NSWorkspace.activeSpaceDidChangeNotification, object: nil)
    }

    private func names() -> [String: String] { (try? NameStore())?.names ?? [:] }

    @objc func refreshTitle() {
        guard let s = SystemSpaces.current().first(where: \.isActive) else { return }
        item.button?.title = names()[s.uuid] ?? "デスクトップ \(s.index)"
    }

    func menuNeedsUpdate(_ menu: NSMenu) {
        menu.removeAllItems()
        let n = names()
        var lastDisplay: String?
        var displayNumber = 0
        for s in SystemSpaces.current() {
            if s.displayID != lastDisplay {
                if lastDisplay != nil { menu.addItem(.separator()) }
                displayNumber += 1
                let header = NSMenuItem(title: "ディスプレイ \(displayNumber)", action: nil, keyEquivalent: "")
                header.isEnabled = false
                menu.addItem(header)
                lastDisplay = s.displayID
            }
            let title = n[s.uuid] ?? (s.isFullscreen ? "フルスクリーン" : "デスクトップ \(s.index)")
            let i = NSMenuItem(title: title, action: #selector(go(_:)), keyEquivalent: "")
            i.target = self
            i.representedObject = s.uuid
            i.state = s.isActive ? .on : .off
            menu.addItem(i)
        }
        menu.addItem(.separator())
        let rename = NSMenuItem(title: "現在のデスクトップの名前を変更…", action: #selector(rename), keyEquivalent: "")
        rename.target = self
        menu.addItem(rename)
        let quit = NSMenuItem(title: "終了", action: #selector(NSApplication.terminate(_:)), keyEquivalent: "q")
        menu.addItem(quit)
    }

    @objc func go(_ sender: NSMenuItem) {
        guard let uuid = sender.representedObject as? String,
              let s = SystemSpaces.current().first(where: { $0.uuid == uuid }) else { return }
        do { try SpaceSwitcher.go(to: s) } catch { show(error) }
    }

    @objc func rename() {
        guard let s = SystemSpaces.current().first(where: \.isActive) else { return }
        if s.isFullscreen { show(DeskmanError.fullscreenNotNameable); return }
        let alert = NSAlert()
        alert.messageText = "デスクトップの名前"
        alert.informativeText = "空にすると名前を解除します。"
        alert.addButton(withTitle: "保存")
        alert.addButton(withTitle: "キャンセル")
        let field = NSTextField(frame: NSRect(x: 0, y: 0, width: 240, height: 24))
        field.stringValue = names()[s.uuid] ?? ""
        alert.accessoryView = field
        NSApp.activate(ignoringOtherApps: true)
        guard alert.runModal() == .alertFirstButtonReturn else { return }
        do {
            var store = try NameStore()
            try store.set(name: field.stringValue, for: s.uuid)
            refreshTitle()
        } catch { show(error) }
    }

    private func show(_ error: Error) {
        let alert = NSAlert()
        alert.messageText = (error as? DeskmanError).map(message(for:)) ?? "\(error)"
        NSApp.activate(ignoringOtherApps: true)
        alert.runModal()
    }
}

@MainActor
func runMenuBar() -> Never {
    let app = NSApplication.shared
    app.setActivationPolicy(.accessory)
    let controller = MenuBarController()
    withExtendedLifetime(controller) { app.run() }
    exit(0)
}
```
